\section*{Phụ lục A: File mã nguồn chính \texttt{tempsen.c}}

#define F_CPU 8000000UL
#include <avr/io.h>
#include <util/delay.h>
#include <stdio.h>
#include "hunget_lcd.h" // Thư viện LCD chuẩn Kit BK

// =======================================================
// CẤU HÌNH CHÂN DHT11 NỐI VÀO CHÂN PA1 TRÊN PORT A
// =======================================================
#define DHT_DDR   DDRA
#define DHT_PORT  PORTA
#define DHT_PIN_R PINA
#define DHT_BIT   PA1

// Hàm đọc dữ liệu từ cảm biến số DHT11
// Trả về 0 nếu thành công, < 0 nếu lỗi/timeout
int8_t DHT11_Read(uint8_t *temp_int, uint8_t *temp_dec, uint8_t *humi_int) {
    uint8_t data[5] = {0, 0, 0, 0, 0};
    uint16_t timeout;

    // 1. MCU gửi tín hiệu Start (kéo LOW tối thiểu 18ms)
    DHT_DDR |= (1 << DHT_BIT);     // Set PA1 là Output
    DHT_PORT &= ~(1 << DHT_BIT);   // Kéo xuống LOW
    _delay_ms(20);                 // Giữ mức LOW 20ms
    DHT_PORT |= (1 << DHT_BIT);    // Kéo lên HIGH
    _delay_us(30);

    // 2. Chuyển chân sang Input để nhận phản hồi từ DHT11
    DHT_DDR &= ~(1 << DHT_BIT);
    DHT_PORT |= (1 << DHT_BIT);    // Bật điện trở kéo lên nội

    // Đợi DHT11 kéo LOW (khoảng 20-40µs)
    timeout = 10000;
    while (DHT_PIN_R & (1 << DHT_BIT)) {
        if (--timeout == 0) return -1; // Timeout chờ DHT11 phản hồi
    }

    // Đợi DHT11 kéo HIGH (khoảng 80µs)
    timeout = 10000;
    while (!(DHT_PIN_R & (1 << DHT_BIT))) {
        if (--timeout == 0) return -2;
    }

    // Đợi DHT11 kéo LOW chuẩn bị truyền bit đầu tiên (khoảng 80µs)
    timeout = 10000;
    while (DHT_PIN_R & (1 << DHT_BIT)) {
        if (--timeout == 0) return -3;
    }

    // 3. Đọc 40 bit dữ liệu (5 byte)
    for (uint8_t i = 0; i < 5; i++) {
        for (int8_t j = 7; j >= 0; j--) {
            // Đợi hết xung LOW 50µs của từng bit
            timeout = 10000;
            while (!(DHT_PIN_R & (1 << DHT_BIT))) {
                if (--timeout == 0) return -4;
            }

            // Đo độ rộng xung HIGH:
            // 26-28µs -> bit 0
            // 70µs    -> bit 1
            // Chờ 38µs rồi đọc trạng thái chân
            _delay_us(38);
            if (DHT_PIN_R & (1 << DHT_BIT)) {
                data[i] |= (1 << j);
                // Đợi xung HIGH kết thúc
                timeout = 10000;
                while (DHT_PIN_R & (1 << DHT_BIT)) {
                    if (--timeout == 0) return -5;
                }
            }
        }
    }

    // 4. Kiểm tra mã kiểm tra lỗi (Checksum)
    if ((uint8_t)(data[0] + data[1] + data[2] + data[3]) != data[4]) {
        return -6; // Sai checksum
    }

    *humi_int  = data[0];
    *temp_int  = data[2];
    *temp_dec  = data[3];
    return 0; // Đọc thành công
}

int main(void) {
    // 1. TẮT JTAG (BẮT BUỘC để giải phóng PORT C điều khiển LCD)
    MCUCSR = (1 << JTD);
    MCUCSR = (1 << JTD);

    // 2. KHỞI TẠO LCD
    LCD4_INIT(0, 0);

    // 3. KHỞI TẠO DÃY LED CẢNH BÁO (PD0 - PD4)
    // PD5-PD7 là Output cho LCD (đã set trong thư viện LCD)
    // Set thêm PD0-PD4 là Output cho LED
    DDRD |= 0x1F;  // Set 5 chân thấp là Output
    PORTD |= 0x1F; // Xuất mức 1 để TẮT hết LED ban đầu (Active Low)

    // --- MÀN HÌNH CHÀO MỪNG ---
    LCD4_CUR_GOTO(1, 0);
    LCD4_OUT_STR("DA2: DO NHIET DO");
    LCD4_CUR_GOTO(2, 2);
    LCD4_OUT_STR("DHT11 SENSOR");
    _delay_ms(1500);
    LCD4_OUT_CMD(0x01); // Xóa màn hình
    _delay_ms(2);

    char buffer[32];
    uint8_t temp_int = 0;
    uint8_t temp_dec = 0;
    uint8_t humi_int = 0;

    while (1) {
        int8_t status = DHT11_Read(&temp_int, &temp_dec, &humi_int);

        if (status == 0) {
            // --- ĐỌC THÀNH CÔNG TỪ DHT11 ---
            // Dòng 1: "Temp: 31.5 °C" (16 ký tự chuẩn)
            LCD4_CUR_GOTO(1, 0);
            sprintf(buffer, "Temp: %u.%u \xDF""C    ", temp_int, temp_dec);
            LCD4_OUT_STR(buffer);

            // Dòng 2: Hiển thị ngưỡng cảnh báo "HIGH Th: 30°C"
            LCD4_CUR_GOTO(2, 0);
            if (temp_int >= 30) {
                // Nhiệt độ vượt ngưỡng 30 độ C: Báo WARNING và BẬT LED (đúng 16 ký tự)
                LCD4_OUT_STR("HIGH Th: 30\xDF""C *!");
                PORTD &= 0xE0; // Bật sáng 5 LED cảnh báo (Active LOW)
            } else {
                // Nhiệt độ bình thường <= 30 độ C: TẮT LED (đúng 16 ký tự)
                LCD4_OUT_STR("HIGH Th: 30\xDF""C    ");
                PORTD |= 0x1F; // Tắt 5 LED cảnh báo
            }
        } else if (status == -6) {
            // Lỗi kiểm tra toàn vẹn dữ liệu (Checksum Error)
            LCD4_CUR_GOTO(1, 0);
            LCD4_OUT_STR("DHT11: CRC Error");
            LCD4_CUR_GOTO(2, 0);
            LCD4_OUT_STR("Checksum Invalid");
            PORTD |= 0x1F; // Cơ chế an toàn (Fail-safe): Tắt LED cảnh báo
        } else {
            // Trường hợp mất tín hiệu, chưa cắm chân PA1 hoặc lỏng dây
            LCD4_CUR_GOTO(1, 0);
            LCD4_OUT_STR("DHT11: No Signal");
            LCD4_CUR_GOTO(2, 0);
            LCD4_OUT_STR("Check Pin PA1   ");
            PORTD |= 0x1F; // Cơ chế an toàn (Fail-safe): Tắt LED cảnh báo
        }

        // Chu kỳ lấy mẫu khuyến nghị theo datasheet của Aosong (2 giây)
        _delay_ms(2000);
    }

    return 0;
}

\section*{Phụ lục B: Thư viện điều khiển LCD \texttt{hunget\_lcd.h}}

\begin{footnotesize}
\begin{verbatim}
#ifndef HUNGET_LCD_H
#define HUNGET_LCD_H

#include <avr/io.h>
#include <util/delay.h>
#include <stdio.h>
#include <stdlib.h>

// =======================================================
// CẤU HÌNH CHÂN THEO ĐÚNG SƠ ĐỒ MẠCH KIT
// =======================================================
#define LCD_DATA_DDR  DDRC
#define LCD_DATA_PORT PORTC
#define LCD_DATA_PIN  PINC

#define LCD_CTRL_DDR  DDRD
#define LCD_CTRL_PORT PORTD

#define LCD_RW 5 // Pin 5 LCD nối PD5
#define LCD_RS 6 // Pin 4 LCD nối PD6
#define LCD_EN 7 // Pin 6 LCD nối PD7

static inline void LCD_Enable(void) {
    LCD_CTRL_PORT |= (1 << LCD_EN);
    _delay_us(5);
    LCD_CTRL_PORT &= ~(1 << LCD_EN);
    _delay_us(100);
}

static inline void LCD_Send4Bit(unsigned char Data) {
    unsigned char current_port_c = LCD_DATA_PORT & 0x0F;
    LCD_DATA_PORT = current_port_c | (Data & 0xF0);
    LCD_Enable();
}

static inline void LCD4_OUT_CMD(unsigned char command) {
    LCD_CTRL_PORT &= ~(1 << LCD_RS); 
    LCD_CTRL_PORT &= ~(1 << LCD_RW);
    LCD_Send4Bit(command);
    LCD_Send4Bit(command << 4);
    if (command < 4) _delay_ms(2);
    else _delay_us(50);
}

static inline void LCD4_OUT_DATA(unsigned char data) {
    LCD_CTRL_PORT |= (1 << LCD_RS);
    LCD_CTRL_PORT &= ~(1 << LCD_RW);
    LCD_Send4Bit(data); 
    LCD_Send4Bit(data << 4); 
    _delay_us(50);
}

static inline void LCD4_INIT(unsigned char cursor, unsigned char blink) {
    LCD_DATA_DDR |= 0xF0; 
    LCD_CTRL_DDR |= (1<<LCD_RS) | (1<<LCD_RW) | (1<<LCD_EN); 
    LCD_CTRL_PORT &= ~((1<<LCD_RS) | (1<<LCD_RW) | (1<<LCD_EN));
    _delay_ms(50);

    LCD_Send4Bit(0x30); _delay_ms(5);
    LCD_Send4Bit(0x30); _delay_ms(2);
    LCD_Send4Bit(0x30); _delay_ms(2);

    LCD_Send4Bit(0x20); _delay_ms(2);

    LCD4_OUT_CMD(0x28); _delay_ms(1);
    LCD4_OUT_CMD(0x08); _delay_ms(1);
    LCD4_OUT_CMD(0x01); _delay_ms(3);

    unsigned char cmd = 0x0C;
    if(cursor) cmd |= 0x02;
    if(blink) cmd |= 0x01;
    LCD4_OUT_CMD(cmd); 
    _delay_ms(1);
    LCD4_OUT_CMD(0x06);
    _delay_ms(1);
}

void LCD4_CUR_GOTO(unsigned char row, unsigned char col) {
    unsigned char address;
    if (row == 1) address = 0x80 + col;
    else address = 0xC0 + col;
    LCD4_OUT_CMD(address);
}

void LCD4_OUT_STR(char *str) {
    while (*str) LCD4_OUT_DATA(*str++);
}

void LCD4_OUT_DEC(unsigned int num, unsigned char len) {
    char buffer[10];
    sprintf(buffer, "%05u", num);
    char *ptr = &buffer[5 - len];
    LCD4_OUT_STR(ptr);
}

void LCD4_DIS_SHIFT(unsigned char direct, unsigned char step) {
    unsigned char i;
    if (direct == 0)
        for(i=0; i<step; i++) LCD4_OUT_CMD(0x18);
    else
        for(i=0; i<step; i++) LCD4_OUT_CMD(0x1C);
}

#endif // HUNGET_LCD_H
\end{verbatim}
\end{footnotesize}
